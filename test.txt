# Run from the project root in PowerShell. Pick a new cache subdirectory for a cold run.
# Terminal 1
npm ci
npm start -- -h 127.0.0.1 -p 3000 -c cache/curl-demo

# Terminal 2. A fresh directory should be absent before the first request.
$base = 'http://127.0.0.1:3000'
$cacheDir = 'cache/curl-demo'
$format = '\nHTTP %{http_code} total %{time_total}s\n'
Test-Path -LiteralPath $cacheDir
Test-Path -LiteralPath 'fixtures/my_image.jpg' -PathType Leaf

# GET from http.cat and then from disk. Expect 200 with different sources and equal hashes.
curl -D - -o NUL -w $format "$base/451"
curl -D - -o NUL -w $format "$base/451"

# Binary PUT and read-back. The fixture path is relative to the current directory.
curl -i -X PUT -H 'Content-Type: image/jpeg' --data-binary '@fixtures/my_image.jpg' -w $format "$base/451"
curl -D - -o NUL -w $format "$base/451"

# Empty PUT must leave the previous JPEG in place.
curl -i -X PUT -w $format "$base/451"
curl -D - -o NUL -w $format "$base/451"

# DELETE existing and absent files, then fetch again. Expected 200, 404, and 200.
curl -i -X DELETE -w $format "$base/451"
Test-Path -LiteralPath "$cacheDir/451.jpg"
curl -i -X DELETE -w $format "$base/451"
curl -D - -o NUL -w $format "$base/451"

# Missing upstream image, invalid route, and unsupported methods.
curl -i -w $format "$base/299"
Test-Path -LiteralPath "$cacheDir/299.jpg"
curl -i -w $format "$base/451/extra"
curl -i -X POST -w $format "$base/451"
curl -I -w $format "$base/451"
curl -i -X POST -w $format "$base/"

# Simulate a local filesystem error on a dedicated test path. Expect 500.
New-Item -ItemType Directory -Path "$cacheDir/500.jpg"
curl -i -w $format "$base/500"
Remove-Item -LiteralPath "$cacheDir/500.jpg"

# Reset the cached image and check a GET with query parameters.
curl -i -X DELETE -w $format "$base/451"
curl -D - -o NUL -w $format "$base/451?demo=1"

# Direct network check used before running the proxy tests.
curl -I https://http.cat/451
